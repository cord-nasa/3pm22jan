<?php 
if (isset($_POST['submit'])) { 
$name = $_POST['name']; 
$bsalary = $_POST['salary']; 
$desig = $_POST['desig']; 
// HRA = 25% of basic 
$hra = ($bsalary * 25) / 100; 
// Default allowances 
$conveyance = 0; 
$extra = 0; 
// Allowances based on designation 
if ($desig == "manager") { 
$conveyance = 1000; 
$extra = 500; 
} elseif ($desig == "supervisor") { 
$conveyance = 750; 
$extra = 200; 
} elseif ($desig == "clerk") { 
$conveyance = 500; 
$extra = 100; 
} elseif ($desig == "peon") { 
$conveyance = 200; 
$extra = 0; 
} 
// Gross Salary 
$gross = $bsalary + $hra + $conveyance + $extra; 
// Income tax 
$incometax = 0; 
if ($gross <= 2000) { 
$incometax = 0; 
} elseif ($gross > 2000 && $gross <= 4000) { 
$incometax = ($gross * 3) / 100; 
} elseif ($gross > 4000 && $gross <= 5000) { 
$incometax = ($gross * 5) / 100; 
} elseif ($gross > 5000) { 
$incometax = ($gross * 8) / 100; 
} 
// Net Salary 
$net = $gross - $incometax; 
// Display Payslip 
echo "<h2>Payslip</h2>"; 
echo "-------------------------<br>"; 
echo "Name: $name <br>"; 
echo "Designation: $desig <br>"; 
echo "Basic Salary: $bsalary <br>"; 
echo "HRA: $hra <br>"; 
echo "Conveyance: $conveyance <br>"; 
echo "Extra Allowance: $extra <br>"; 
echo "Gross Salary: $gross <br>"; 
echo "Income Tax: $incometax <br>"; 
echo "Net Salary: $net <br>"; 
echo "-------------------------<br>"; 
}
